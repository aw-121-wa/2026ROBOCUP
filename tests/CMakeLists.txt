cmake_minimum_required(VERSION 3.22)
project(chassis_tests C)
enable_testing()
# Keep object paths ASCII for Windows MinGW assemblers.
configure_file(../Algorithm/motion.c ${CMAKE_CURRENT_BINARY_DIR}/motion.c COPYONLY)
configure_file(../DRIVER/src/zdt_x42s.c ${CMAKE_CURRENT_BINARY_DIR}/zdt_x42s.c COPYONLY)
configure_file(../Algorithm/planner.c ${CMAKE_CURRENT_BINARY_DIR}/planner.c COPYONLY)
configure_file(../Algorithm/imu_health.c ${CMAKE_CURRENT_BINARY_DIR}/imu_health.c COPYONLY)
add_executable(test_motion test_motion.c ${CMAKE_CURRENT_BINARY_DIR}/motion.c ${CMAKE_CURRENT_BINARY_DIR}/planner.c ${CMAKE_CURRENT_BINARY_DIR}/zdt_x42s.c)
add_executable(test_progress test_progress.c ${CMAKE_CURRENT_BINARY_DIR}/planner.c)
add_executable(test_health test_health.c ${CMAKE_CURRENT_BINARY_DIR}/imu_health.c)
foreach(test_target test_progress test_health)
    target_include_directories(${test_target} PRIVATE ../Algorithm)
    target_compile_options(${test_target} PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(${test_target} PRIVATE m)
    add_test(NAME ${test_target} COMMAND ${test_target})
endforeach()
target_include_directories(test_motion PRIVATE ../Algorithm ../DRIVER/inc)
target_compile_options(test_motion PRIVATE -Wall -Wextra -Werror)
target_link_libraries(test_motion PRIVATE m)
add_test(NAME motion COMMAND test_motion)
configure_file(../Algorithm/heading.c ${CMAKE_CURRENT_BINARY_DIR}/heading.c COPYONLY)
configure_file(../App/chassis_odom.c ${CMAKE_CURRENT_BINARY_DIR}/chassis_odom.c COPYONLY)
add_executable(test_control_math test_control_math.c ${CMAKE_CURRENT_BINARY_DIR}/heading.c
    ${CMAKE_CURRENT_BINARY_DIR}/chassis_odom.c ${CMAKE_CURRENT_BINARY_DIR}/motion.c)
target_include_directories(test_control_math PRIVATE ../Algorithm ../App)
target_compile_options(test_control_math PRIVATE -Wall -Wextra -Werror)
target_link_libraries(test_control_math PRIVATE m)
add_test(NAME control_math COMMAND test_control_math)
